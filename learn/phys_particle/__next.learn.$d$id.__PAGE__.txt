const dt = 0.05;
const ground = -150;
const r = 7;                  // 粒子半径
const colors = ["#F59E0B", "#38bdf8", "#E2E8F0", "#22C55E"];

const px = [];                // 每个粒子的横坐标
const py = [];                // 每个粒子的纵坐标
const vx = [];                // 每个粒子的水平速度
const vy = [];                // 每个粒子的竖直速度
for (let i = 0; i < N; i++) { // 初始化：给每个粒子一个不同的起点和速度
  px.push(-190 + i * 34);
  py.push(110 + (i % 3) * 28);
  vx.push((i % 2 === 0 ? 1 : -1) * (25 + i * 3));
  vy.push(0);
}

for (let frame = 0; frame < 50; frame++) {
  for (let i = 0; i < N; i++) {          // 挨个更新每个粒子
    vy[i] = vy[i] - g * dt;              // 重力
    px[i] = px[i] + vx[i] * dt;
    py[i] = py[i] + vy[i] * dt;
    if (py[i] < ground + r) {            // 撞到地面：反弹并损失能量
      py[i] = ground + r;
      vy[i] = -vy[i] * 0.6;
    }
    if (px[i] < -230 + r) {              // 撞到左墙：水平速度反向
      px[i] = -230 + r;
      vx[i] = -vx[i];
    }
    if (px[i] > 230 - r) {               // 撞到右墙
      px[i] = 230 - r;
      vx[i] = -vx[i];
    }
  }

  __runtime.clearCanvas();
  __runtime.drawRect(-240, ground - 40, 480, 40, "#334155");     // 地面
  for (let i = 0; i < N; i++) {
    __runtime.drawCircle(px[i], py[i], r, colors[i % colors.length]);
  }
  __runtime.wait(dt);
