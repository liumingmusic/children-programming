const ground = -150;   // 地面的高度
const r = 12;          // 小球半径
let y = 80;            // 小球高度
let v = 0;             // 速度（往下为负）
const g = 800;         // 重力加速度
const dt = 0.05;       // 每一帧走过的时间（秒）
let bounces = 0;       // 弹了几次

for (let frame = 0; frame < 50; frame++) {
  v = v - g * dt;      // 重力拉低速度
  y = y + v * dt;      // 位置跟着速度走

  if (y < ground + r && v < 0) {   // 球心低于「地面 + 半径」= 撞地了
    y = ground + r;                // 拉回地面，别陷进去
    if (-v > 60) {                 // 还弹得动 → 反弹
      v = -v * 0.7;                // 速度反向，只保留 70% 能量
      bounces = bounces + 1;
    } else {
      v = 0;                       // 弹不动了，安静地停在地面
    }
  }

  __runtime.clearCanvas();                                        // 擦掉上一帧
  __runtime.drawRect(-240, ground - 40, 480, 40, "#334155");       // 地面
  __runtime.drawCircle(0, y, r, "#F59E0B");                        // 小球
  __runtime.drawText(-225, 155, "弹跳次数：" + bounces, "#E2E8F0", 16);
  __runtime.wait(dt);                                             // 停一下才看得见
}
