  const s = scores[i];
  let k = 0;
  if (s >= 90) k = 4;                           // 从高往低比，顺序不能反
  else if (s >= 80) k = 3;
  else if (s >= 70) k = 2;
  else if (s >= 60) k = 1;
  else k = 0;
  counts[k] = counts[k] + 1;                    // 落进哪个桶，哪个桶 +1
}

for (let i = 0; i < counts.length; i++) {
